2. Pointers — The Biggest Difference From Java

int a = 42;
int *p = &a;     // p stores ADDRESS of a; & = address-of
*p               // dereference → 42
printf("%p",&a); // address where a lives

int *p = NULL;   // null pointer (points to 0)
int *r;          // UNINITIALIZED — dereferencing = UB/segfault

Pointer arithmetic (scales by sizeof(type))

int arr[5] = {10,20,30,40,50};
int *p = arr;         // p → arr[0]
*(p+1)                // 20 (adds 1×sizeof(int)=4 bytes)
p[2] == *(p+2)        // TRUE — array notation works on pointers
*arr + 1              // 11 (NOT arr[1])
q - p                 // ELEMENT COUNT, not bytes

Void, function, double pointers

void *ptr;            // generic — must cast before use
double *pp; int **pp; // pointer to pointer (2D arrays, modifying a pointer in a fn)
int (*fp)(int,int); fp = &add; fp(5,3);   // function pointer → callbacks, ISR tables
int (*ops[])(int,int) = {add, sub};      // array of function pointers (lookup table)

3. Arrays & the Decay Trap

int arr[5] = {1,2};        // rest zeroed: {1,2,0,0,0}
int arr3[5] = {0};         // all zeros
int *p = arr;              // arr DECAYS to int* (address of arr[0])

sizeof(arr)                // 20 (5 × 4 bytes — full array)
sizeof(p)                  // 8 (just the pointer on 64-bit)
void f(int a[]) { sizeof(a); }   // 8 — decayed to pointer! ★#1 MCQ trap

int vla[n];                // VLA (C99 only)
int *h = malloc(5*sizeof(int));  // heap array — free later

No bounds checking: arr[10] = 42 on a size-5 array = UB (silent corruption). Java throws; C corrupts.

4. Strings

char s1[] = "Hello";       // 6 bytes: H e l l o \0
char *s3 = "Hello";        // pointer to STRING LITERAL — READ-ONLY!
s1[0] = 'h';               // OK (mutable array)
// s3[0] = 'h';            // UB — literal may live in ROM

sizeof(s1) = 6;            // includes '\0'
strlen(s1)  = 5;           // excludes '\0'
sizeof(s3)  = 8;           // pointer size

strcpy(d,s); strncpy(d,s,n); strcat; strncat;
strcmp(a,b)==0;            // compare content — NEVER use == (compares addresses!)
strstr; strchr; sprintf(buf,"%d",42);

int *p = malloc(10*sizeof(int));   // UNINITIALIZED
int *q = calloc(10, sizeof(int));  // zero-initialized
p = realloc(p, 20*sizeof(int));    // may MOVE; DON'T do p=realloc(p,x) (loses original on fail)
free(p); p = NULL;                 // ALWAYS null after free
free(p);                           // double free = UB ★

6. struct / union / enum / bit-fields

typedef struct { uint8_t h,m,s; } Time;
Time t;  t.h=10;          // dot
Time *tp = &t;  tp->h=11; // arrow

// ★PADDING — HIGH MCQ FREQUENCY
struct S { char c; int i; };  // sizeof = 8, NOT 5 (3 pad bytes after c)
// offset rule: each member at multiple of its size; struct rounded to largest member's alignment

struct P { char a; int b; char c; };  // 12 bytes
struct Q { char a; char c; int b; };  // 8 bytes — reorder largest→smallest to save!

// #pragma pack(1) forces 1-byte alignment for wire/protocol structs (7 bytes above)
// union — members SHARE memory, size = largest member
union U { uint32_t full; uint8_t bytes[4]; };  // sizeof 4
// enum: 0,1,2...
typedef struct { uint32_t en:1, mode:3, rd:1; } Ctrl;  // bit fields

7. Storage Classes — static / volatile / const / extern

  • static local — persists across calls, initialized ONCE. static file-scope — visible only in this .c (information hiding).
  • volatile — compiler MUST NOT optimize reads/writes (hardware registers, ISR flags). while(!flag); without volatile may become an infinite loop.
  • const combos: const int *p (value const, pointer moves) · int *const p (pointer const) · const int *const p (both). int const *s = const int *.
  • extern — defined in another file.
  • Hardware register: volatile uint32_t *reg = (volatile uint32_t*)0x40000000;
  • Classic: a hardware status register is volatile const — changes on its own, but the program must not write it.

8. Preprocessor & Macros

#define PI 3.14159
#define SQUARE(x) x*x        // ★BUG: SQUARE(3+1) = 3+1*3+1 = 7, NOT 16
#define SQUARE(x) ((x)*(x))  // correct
#define MAX(a,b) ((a)>(b)?(a):(b))   // BUT: MAX(x++,4) evaluates x++ twice!

#include <stdio.h>    // system path
#include "myfile.h"   // current dir first
#ifdef DEBUG ... #endif
#ifndef HEADER_H ... #define HEADER_H ... #endif   // header guard
  • Macros: no type checking, no debug info, side effects (args evaluated multiple times). Prefer static inline functions (type-safe, no call overhead).

11. Compilation Model

gcc -c main.c -o main.o     // compile → object
gcc -c utils.c -o utils.o

12. THE C MCQ CHEAT-SHEET (memorize all 15)

  1. 052 is OCTAL = 42 decimal, NOT 52
  2. "Hello" == "Hello" compares ADDRESSES → use strcmp()
  3. sizeof(arr) inside a function = pointer size (8), not array size
  4. if (x = 5) is ASSIGNMENT → always TRUE
  5. ++i vs i++ — pre vs post-increment
  6. #define SQUARE(x) x*x → SQUARE(1+2)=5, not 9
  7. struct { char a; int b; } sizeof = 8 (padding)
  8. free(p); p = NULL; ALWAYS
  9. p = realloc(p, big) — on failure p=NULL, original LOST
  10. '\0' (0) vs '0' (48)
  11. int *p; *p = 5; — uninitialized deref = SEGFAULT
  12. scanf("%d",&x) — & required for primitives
  13. Never compare floats with ==
  14. unsigned int x = -1; → wraps to 4294967295 (UINT_MAX)
  15. a[i] = *(a+i) = i[a] — YES, i[a] is valid C
